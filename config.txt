-- Instalar o Angular Global
npm install -g @angular/cli

-- Resolvendo problemas de execução
npm config get prefix --> adicionar ao PATH do Windows

-- Criar um projeto Angular
ng new nome-projeto --routing=false --style=css --skip-tests --standalone --ssr=false --file-name-style-guide=2025 --defaults

-- Executar 
ng serve --open

-- Para criar componentes (não poder estar executando o projeto)
ng generate component components/nome-component --skip-tests

-- Criar serviço
ng generate service services/nome-servico --skip-tests
